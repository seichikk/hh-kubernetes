
==== rustfs из дз-2 внутри кластера: statefulset с четырьмя томами ====

$ cat k8s/20-rustfs.yaml
apiVersion: v1
kind: Secret
metadata:
  name: s3-keys
  namespace: hh-s3
stringData:
  S3_ACCESS_KEY: rustfsadmin
  S3_SECRET_KEY: rustfsadmin
---
apiVersion: v1
kind: Service
metadata:
  name: rustfs
  namespace: hh-s3
spec:
  selector:
    app: rustfs
  ports:
    - name: s3
      port: 9000
    - name: console
      port: 9001
---
apiVersion: apps/v1
kind: StatefulSet
metadata:
  name: rustfs
  namespace: hh-s3
spec:
  serviceName: rustfs
  replicas: 1
  selector:
    matchLabels:
      app: rustfs
  template:
    metadata:
      labels:
        app: rustfs
    spec:
      containers:
        - name: rustfs
          image: rustfs/rustfs:1.0.1
          env:
            - name: RUSTFS_VOLUMES
              value: /data/rustfs{1...4}
            - name: RUSTFS_ACCESS_KEY
              valueFrom:
                secretKeyRef:
                  name: s3-keys
                  key: S3_ACCESS_KEY
            - name: RUSTFS_SECRET_KEY
              valueFrom:
                secretKeyRef:
                  name: s3-keys
                  key: S3_SECRET_KEY
            - name: RUSTFS_ADDRESS
              value: 0.0.0.0:9000
            - name: RUSTFS_CONSOLE_ENABLE
              value: "true"
            - name: RUSTFS_CONSOLE_ADDRESS
              value: 0.0.0.0:9001
            - name: RUSTFS_UNSAFE_BYPASS_DISK_CHECK
              value: "true"
            - name: RUSTFS_OBS_LOGGER_LEVEL
              value: warn
          ports:
            - name: s3
              containerPort: 9000
            - name: console
              containerPort: 9001
          readinessProbe:
            httpGet:
              path: /health/live
              port: s3
            periodSeconds: 2
          resources:
            requests:
              cpu: 500m
              memory: 1Gi
            limits:
              cpu: "2"
              memory: 2Gi
          volumeMounts:
            - name: disk1
              mountPath: /data/rustfs1
            - name: disk2
              mountPath: /data/rustfs2
            - name: disk3
              mountPath: /data/rustfs3
            - name: disk4
              mountPath: /data/rustfs4
  volumeClaimTemplates:
    - metadata:
        name: disk1
      spec:
        accessModes: [ReadWriteOnce]
        resources:
          requests:
            storage: 1Gi
    - metadata:
        name: disk2
      spec:
        accessModes: [ReadWriteOnce]
        resources:
          requests:
            storage: 1Gi
    - metadata:
        name: disk3
      spec:
        accessModes: [ReadWriteOnce]
        resources:
          requests:
            storage: 1Gi
    - metadata:
        name: disk4
      spec:
        accessModes: [ReadWriteOnce]
        resources:
          requests:
            storage: 1Gi

==== перенос бакета hh-lake из хранилища на хосте ====

$ cat k8s/21-copy.yaml
apiVersion: batch/v1
kind: Job
metadata:
  name: s3-copy
  namespace: hh-s3
spec:
  backoffLimit: 0
  template:
    spec:
      restartPolicy: Never
      containers:
        - name: copy
          image: hh-k8s:1
          args: ["copy", "--prefix", "raw/it_vacancies_2024.csv", "--prefix", "results/mapreduce/"]
          env:
            - name: SRC_ENDPOINT
              value: http://host.docker.internal:19000
            - name: S3_ENDPOINT
              value: http://rustfs:9000
            - name: S3_BUCKET
              value: hh-lake
          envFrom:
            - secretRef:
                name: s3-keys
          resources:
            requests:
              cpu: 500m
              memory: 512Mi
            limits:
              cpu: "1"
              memory: 512Mi
